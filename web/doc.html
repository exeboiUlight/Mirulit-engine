Обзор движка

MIRULIT — это легковесный 2D игровой движок на языке C, построенный на SDL3. Предназначен для быстрой разработки игр с минимальными зависимостями.

Примечание: Движок находится в активной разработке. API может меняться между версиями.

Основные возможности:

  • ✅ Полноценная система сущностей (Entity Component System)
  • ✅ Управление частицами с физикой
  • ✅ 2D камера с зумом, вращением и тряской
  • ✅ Система коллизий AABB
  • ✅ Полная поддержка ввода (клавиатура, мышь)
  • ✅ Управление временем и FPS
  • ✅ Статистика и отладочная информация
  • ✅ Поддержка SDL3 и совместимость с C99

Быстрый старт

Минимальный пример игры:

main.c 
#include "mirulit.h"

MIR_Entity* player = NULL;

void UpdatePlayer(MIR_Entity* entity, float delta) {
    // Движение WASD
    float speed = 200.0f;
    if (MIR_IsKeyDown(SDL_SCANCODE_W))
        entity->transform.position.y -= speed * delta;
    if (MIR_IsKeyDown(SDL_SCANCODE_S))
        entity->transform.position.y += speed * delta;
    if (MIR_IsKeyDown(SDL_SCANCODE_A))
        entity->transform.position.x -= speed * delta;
    if (MIR_IsKeyDown(SDL_SCANCODE_D))
        entity->transform.position.x += speed * delta;
}

int main(int argc, char* argv[]) {
    // Инициализация движка
    if (!MIR_Init("My First Game", 800, 600)) {
        return 1;
    }
    
    // Создание игрока
    player = MIR_CreateEntity("player");
    player->transform.position = (MIR_Vec2){400, 300};
    player->transform.scale = (MIR_Vec2){50, 50};
    player->sprite.color = MIR_COLOR_GREEN;
    player->update = UpdatePlayer;
    
    // Основной игровой цикл
    while (MIR_IsRunning()) {
        MIR_ProcessEvents();
        MIR_BeginFrame();
        
        MIR_UpdateEntities();
        MIR_UpdateParticles();
        MIR_ResolveCollisions();
        
        MIR_DrawEntities();
        MIR_DrawParticles();
        MIR_DrawDebugInfo();
        
        MIR_EndFrame();
    }
    
    MIR_Shutdown();
    return 0;
}

Создание эффекта взрыва 
void CreateExplosion(MIR_Vec2 position, int count) {
    for (int i = 0; i < count; i++) {
        // Случайное направление
        float angle = MIR_Math_RandomRange(0, 2 * M_PI);
        float speed = MIR_Math_RandomRange(100, 400);
        
        MIR_Vec2 velocity = {
            cosf(angle) * speed,
            sinf(angle) * speed
        };
        
        // Случайный цвет огня
        MIR_Color colors[] = {
            MIR_COLOR_RED,
            MIR_COLOR_ORANGE,
            MIR_COLOR_YELLOW,
            {255, 100, 0, 255}  // Темно-оранжевый
        };
        
        MIR_Color color = colors[rand() % 4];
        
        // Размер и время жизни
        float size = MIR_Math_RandomRange(3, 8);
        float life = MIR_Math_RandomRange(0.5, 1.5);
        
        // Создание частицы
        MIR_EmitParticleEx(
            position,
            velocity,
            (MIR_Vec2){0, 200},  // Гравитация вниз
            color,
            size,
            life
        );
    }
}

Камера, следующая за игроком 
// В основном игровом цикле
void UpdateCamera(void) {
    if (player) {
        // Плавное следование за игроком
        MIR_SetCameraTarget(player->transform.position);
        
        // Зум в зависимости от скорости
        float speed = sqrtf(
            player->transform.velocity.x * player->transform.velocity.x +
            player->transform.velocity.y * player->transform.velocity.y
        );
        
        float targetZoom = 1.0f + speed / 500.0f;
        targetZoom = MIR_Math_Clamp(targetZoom, 0.8f, 2.0f);
        
        // Плавный зум
        float currentZoom = _mir->camera.zoom;
        float newZoom = MIR_Math_Lerp(currentZoom, targetZoom, 
                                     _mir->delta_time * 2.0f);
        MIR_SetCameraZoom(newZoom);
        
        // Тряска при повреждении
        if (player_took_damage) {
            MIR_CameraShake(10.0f, 0.3f);
            player_took_damage = false;
        }
    }
}

// Callback функция коллизии
void OnPlayerEnemyCollision(MIR_Entity* player, MIR_Entity* enemy) {
    PlayerData* player_data = (PlayerData*)player->user_data;
    EnemyData* enemy_data = (EnemyData*)enemy->user_data;
    
    // Игрок получает урон
    player_data->health -= enemy_data->damage;
    
    // Враг уничтожается
    enemy->active = false;
    
    // Создаем эффект взрыва
    CreateExplosion(enemy->transform.position, 20);
    
    // Добавляем очки
    player_data->score += enemy_data->points;
    
    // Тряска камеры
    MIR_CameraShake(5.0f, 0.2f);
}

// Настройка коллайдеров при создании сущностей

    enemy->collider.bounds = (MIR_Rect){-15, -15, 30, 30};

// Плавное движение к цели
MIR_Vec2 MoveTowards(MIR_Vec2 current, MIR_Vec2 target, float speed) {
    MIR_Vec2 direction = MIR_Vec2_Subtract(target, current);
    direction = MIR_Math_Normalize(direction);
    return MIR_Vec2_Add(current, MIR_Vec2_Multiply(direction, speed));
}

// Проверка дистанции для AI
bool IsInRange(MIR_Entity* entity, MIR_Vec2 target, float range) {
    float dist = MIR_Math_Distance(entity->transform.position, target);
    return dist <= range;
}

// Плавное появление/исчезновение (fade)
void UpdateFade(MIR_Entity* entity, float target_alpha, float speed) {
    float current = entity->sprite.color.a / 255.0f;
    float new_alpha = MIR_Math_Lerp(current, target_alpha, 
                                   MIR_GetDeltaTime() * speed);
    entity->sprite.color.a = (uint8_t)(new_alpha * 255);
}

// Случайная позиция в круге
MIR_Vec2 RandomPointInCircle(MIR_Vec2 center, float radius) {
    float angle = MIR_Math_RandomRange(0, 2 * M_PI);
    float r = MIR_Math_RandomRange(0, radius);
    
    return (MIR_Vec2){
        center.x + cosf(angle) * r,
        center.y + sinf(angle) * r
    };
}

Space Shooter - основная логика 
// Структура для данных игрока
typedef struct {
    int score;
    int lives;
    float fire_rate;
    float fire_timer;
} PlayerData;

// Структура для данных снаряда
typedef struct {
    float damage;
    float speed;
    MIR_Color trail_color;
} ProjectileData;

void UpdatePlayer(MIR_Entity* player, float delta) {
    PlayerData* data = (PlayerData*)player->user_data;
    
    // Движение
    float speed = 300.0f;
    if (MIR_IsKeyDown(SDL_SCANCODE_W)) player->transform.position.y -= speed * delta;
    if (MIR_IsKeyDown(SDL_SCANCODE_S)) player->transform.position.y += speed * delta;
    if (MIR_IsKeyDown(SDL_SCANCODE_A)) player->transform.position.x -= speed * delta;
    if (MIR_IsKeyDown(SDL_SCANCODE_D)) player->transform.position.x += speed * delta;
    
    // Стрельба
    data->fire_timer += delta;
    if (MIR_IsKeyDown(SDL_SCANCODE_SPACE) && data->fire_timer >= data->fire_rate) {
        CreateProjectile(player->transform.position);
        data->fire_timer = 0;
        
        // Эффект отдачи
        player->transform.position.x -= 5;
        MIR_CameraShake(2.0f, 0.1f);
    }
    
    // Границы экрана
    player->transform.position.x = MIR_Math_Clamp(player->transform.position.x, 50, 750);
    player->transform.position.y = MIR_Math_Clamp(player->transform.position.y, 50, 550);
}

void CreateProjectile(MIR_Vec2 position) {
    MIR_Entity* projectile = MIR_CreateEntity("projectile");

    projectile->transform.scale = (MIR_Vec2){5, 15};

    // Коллайдер

    // Эффект частиц
    for (int i = 0; i < 3; i++) {
        MIR_EmitParticle(
            position,
            (MIR_Vec2){MIR_Math_RandomRange(-20, 20), 50},
            MIR_COLOR_ORANGE,
            MIR_Math_RandomRange(1, 3),
            0.3f
        );
    }
}

Пример 2: Менеджер игры

Управление состоянием игры 
typedef enum {
    GAME_STATE_MENU,
    GAME_STATE_PLAYING,
    GAME_STATE_PAUSED,
    GAME_STATE_GAME_OVER
} GameState;

GameState current_state = GAME_STATE_MENU;
int score = 0;
int high_score = 0;

void UpdateGameState(void) {
    switch (current_state) {
        case GAME_STATE_MENU:
            UpdateMenu();
            if (MIR_IsKeyPressed(SDL_SCANCODE_RETURN)) {
                StartNewGame();
                current_state = GAME_STATE_PLAYING;
            }
            break;
            
        case GAME_STATE_PLAYING:
            UpdateGame();
            if (MIR_IsKeyPressed(SDL_SCANCODE_ESCAPE)) {
                current_state = GAME_STATE_PAUSED;
                MIR_Pause();
            }
            if (IsGameOver()) {
                current_state = GAME_STATE_GAME_OVER;
                if (score > high_score) high_score = score;
            }
            break;
            
        case GAME_STATE_PAUSED:
            if (MIR_IsKeyPressed(SDL_SCANCODE_ESCAPE)) {
                current_state = GAME_STATE_PLAYING;
                MIR_Resume();
            }
            if (MIR_IsKeyPressed(SDL_SCANCODE_M)) {
                current_state = GAME_STATE_MENU;
                CleanupGame();
                MIR_Resume();
            }
            break;
            
        case GAME_STATE_GAME_OVER:
            UpdateGameOver();
            if (MIR_IsKeyPressed(SDL_SCANCODE_R)) {
                StartNewGame();
                current_state = GAME_STATE_PLAYING;
            }
            break;
    }
}

void DrawUI(void) {
    // Отрисовка счета
    char score_text[64];
    snprintf(score_text, sizeof(score_text), "Score: %d", score);
    DrawText(score_text, 10, 10, MIR_COLOR_WHITE);
    
    // Отрисовка жизней
    for (int i = 0; i < player_lives; i++) {
        MIR_Rect heart = {10 + i * 30, 40, 25, 25};
        MIR_DrawRect(heart, MIR_COLOR_RED);
    }
    
    // Состояние игры
    if (current_state == GAME_STATE_PAUSED) {
        DrawCenteredText("PAUSED", MIR_COLOR_YELLOW);
        DrawCenteredText("Press ESC to resume", MIR_COLOR_WHITE);
    }
    
    if (current_state == GAME_STATE_GAME_OVER) {
        DrawCenteredText("GAME OVER", MIR_COLOR_RED);
        char hs_text[64];
        snprintf(hs_text, sizeof(hs_text), "High Score: %d", high_score);
        DrawCenteredText(hs_text, MIR_COLOR_YELLOW);
        DrawCenteredText("Press R to restart", MIR_COLOR_WHITE);
    }
}
Важно: Не забывайте освобождать память при уничтожении сущностей. Используйте on_destroy callback для очистки user_data.
Производительность: Для больших проектов ограничьте количество сущностей и частиц, используйте пулы объектов и избегайте частых аллокаций памяти в основном цикле.

Установка и использование

Требования:

  • Компилятор C (GCC, Clang, MSVC)
  • SDL3 библиотека
  • CMake (опционально)

Сборка:

Сборка движка 
# Клонирование репозитория
git clone https://github.com/yourusername/mirulit.git
cd mirulit

# Сборка библиотеки
mkdir build && cd build
cmake ..
make

# Установка (опционально)
sudo make install

# Использование в своем проекте
cp mirulit.h your_project/
cp libmirulit.a your_project/
